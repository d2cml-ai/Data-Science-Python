import rioxarray as rxr
import xarray as xr
from osgeo import gdal
import os

# Lista de archivos .hdf
tiles = [
    "MODIS_tiles/h10v09.hdf",
    "MODIS_tiles/h10v10.hdf",
    "MODIS_tiles/h11v09.hdf",
    "MODIS_tiles/h11v10.hdf"
]

# Extraer los subdatasets a GeoTIFFs temporales
def get_percent_tree_cover_path(hdf_file):
    ds = gdal.Open(hdf_file)
    subs = ds.GetSubDatasets()
    for s in subs:
        if "Percent_Tree_Cover" in s[0]:
            return s[0]
    return None

output_files = []

for i, f in enumerate(tiles):
    path = get_percent_tree_cover_path(f)
    if path:
        outname = f"tile_{i}.tif"
        gdal.Translate(outname, path)
        output_files.append(outname)
        print(f"✔️ Exportado {f} → {outname}")
    else:
        print(f"❌ No se encontró Percent_Tree_Cover en {f}")

# Leer los .tif y crear mosaico
dataarrays = [rxr.open_rasterio(f, masked=True) for f in output_files]

if len(dataarrays) == 0:
    raise ValueError("❌ No se cargó ningún archivo válido.")

mosaic = xr.concat(dataarrays, dim="x")
mosaic = mosaic.rio.reproject("EPSG:4326")
mosaic.rio.to_raster("vcf_2024_peru.tif")

print("✅ Raster guardado como vcf_2024_peru.tif")

import geopandas as gpd
from rasterstats import zonal_stats
import pandas as pd
from shapely.geometry import Polygon

# 1. Cargar las áreas protegidas
gdf = gpd.read_file("peru_protected_areas.gpkg")
gdf = gdf.to_crs(epsg=32718)  # Usamos UTM zona 18S (metros) para trabajar en km correctamente

# Aseguramos un ID para cada parque
gdf["park_id"] = gdf.index

# 2. Buffer distances (en metros)
buffers_km = [5, 10, 20, 25]
buffer_m = [b * 1000 for b in buffers_km]

# 3. Leer el raster
raster_path = "vcf_2024_peru.tif"

# 4. Almacenar resultados
results = []

for b in buffer_m:
    # Zonas interiores (dentro del parque)
    gdf_inner = gdf.copy()
    gdf_inner["geometry"] = gdf_inner.geometry.buffer(-b)
    gdf_inner = gdf_inner[gdf_inner.is_valid & ~gdf_inner.is_empty]

    zs_inner = zonal_stats(
        gdf_inner, raster_path,
        stats=["mean", "median", "std", "count"],
        nodata=255
    )

    for i, stat in enumerate(zs_inner):
        results.append({
            "park_id": gdf_inner.iloc[i]["park_id"],
            "buffer_km": b / 1000,
            "in_out": "inner",
            "mean_tc": stat.get("mean"),
            "median_tc": stat.get("median"),
            "sd_tc": stat.get("std"),
            "n_pixels": stat.get("count")
        })

    # Zonas exteriores (fuera del parque)
    gdf_outer = gdf.copy()
    gdf_outer["geometry"] = gdf_outer.geometry.buffer(b)
    gdf_outer = gpd.overlay(gdf_outer, gdf, how="difference")  # quitamos el área original

    zs_outer = zonal_stats(
        gdf_outer, raster_path,
        stats=["mean", "median", "std", "count"],
        nodata=255
    )

    for i, stat in enumerate(zs_outer):
        results.append({
            "park_id": gdf_outer.iloc[i]["park_id"],
            "buffer_km": b / 1000,
            "in_out": "outer",
            "mean_tc": stat.get("mean"),
            "median_tc": stat.get("median"),
            "sd_tc": stat.get("std"),
            "n_pixels": stat.get("count")
        })

# 5. Guardar resultados
df_results = pd.DataFrame(results)
df_results.to_csv("zonal_stats_peru.csv", index=False)
print("✅ Listo. Resultados guardados en zonal_stats_peru.csv")



import geopandas as gpd

# Cargar y preparar
gdf = gpd.read_file("peru_protected_areas.gpkg")
gdf = gdf.to_crs(epsg=32718)
gdf["park_id"] = gdf.index
gdf_sample = gdf.head(1).copy()  # solo el primer parque

raster_path = "vcf_2024_peru.tif"
buffer_kms = [5, 10]
buffer_m = [b * 1000 for b in buffer_kms]
results = []

for b in buffer_m:
    # INNER buffer
    gdf_inner = gdf_sample.copy()
    gdf_inner["geometry"] = gdf_inner.geometry.buffer(-b)
    gdf_inner = gdf_inner[gdf_inner.is_valid & ~gdf_inner.is_empty]

    zs_inner = zonal_stats(
        gdf_inner, raster_path,
        stats=["mean", "median", "std", "count"],
        nodata=255
    )

    for i, stat in enumerate(zs_inner):
        results.append({
            "park_id": gdf_inner.iloc[i]["park_id"],
            "buffer_km": b / 1000,
            "in_out": "inner",
            "mean_tc": stat.get("mean"),
            "median_tc": stat.get("median"),
            "sd_tc": stat.get("std"),
            "n_pixels": stat.get("count")
        })

    # OUTER buffer
    gdf_outer = gdf_sample.copy()
    gdf_outer["geometry"] = gdf_outer.geometry.buffer(b)
    gdf_outer = gpd.overlay(gdf_outer, gdf_sample, how="difference")

    zs_outer = zonal_stats(
        gdf_outer, raster_path,
        stats=["mean", "median", "std", "count"],
        nodata=255
    )

    for i, stat in enumerate(zs_outer):
        results.append({
            "park_id": gdf_outer.iloc[i]["park_id"],
            "buffer_km": b / 1000,
            "in_out": "outer",
            "mean_tc": stat.get("mean"),
            "median_tc": stat.get("median"),
            "sd_tc": stat.get("std"),
            "n_pixels": stat.get("count")
        })

# Guardar resultados
df_results = pd.DataFrame(results)
df_results.to_csv("zonal_stats_1park.csv", index=False)
